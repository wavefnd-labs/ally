cmake_minimum_required(VERSION 3.25)
project(Ally VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
find_package(Threads REQUIRED)
if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Linux|Windows|Darwin)$")
    message(FATAL_ERROR "Ally supports Linux, Windows and macOS desktops only.")
endif()
option(ALLY_BUILD_APP "Build SDL3 + Skia workstation" ON)
option(ALLY_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(ALLY_SANITIZERS AND NOT MSVC)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_subdirectory(libs/media)
add_subdirectory(libs/audio)
add_library(ally_core src/project/project.cpp src/midi/model.cpp)
target_include_directories(ally_core PUBLIC include)
target_link_libraries(ally_core PUBLIC Ally::Audio Ally::Media)
set(ally_targets ally_core ally_audio ally_media)
if(ALLY_BUILD_APP)
    include(cmake/Dependencies.cmake)
    add_executable(ally src/app/main.cpp src/graphics/renderer.cpp src/ui/workspace/workspace.cpp src/midi/input.cpp)
    target_include_directories(ally PRIVATE include)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT ALLY_LIBDECOR_FOUND)
        target_compile_definitions(ally PRIVATE ALLY_PREFER_X11_WITHOUT_LIBDECOR)
        message(STATUS "libdecor development files missing: prefer X11 native window decorations.")
    endif()
    target_link_libraries(ally PRIVATE ally_core SDL3::SDL3 Skia::Skia libremidi::libremidi)
    list(APPEND ally_targets ally)
    install(TARGETS ally RUNTIME DESTINATION bin)
endif()
foreach(target IN LISTS ally_targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
if(BUILD_TESTING)
    add_executable(project_tests tests/project_tests.cpp)
    target_link_libraries(project_tests PRIVATE ally_core)
    add_test(NAME ally.project COMMAND project_tests)
endif()
install(FILES README.md DESTINATION share/ally)
install(DIRECTORY assets/ DESTINATION share/ally/assets)
install(DIRECTORY docs/ DESTINATION share/ally/docs)
set(CPACK_PACKAGE_NAME "ally")
set(CPACK_PACKAGE_VERSION "0.1.0")
set(CPACK_GENERATOR "ZIP")
include(CPack)
