cmake_minimum_required(VERSION 3.25)
project(AllyAudio VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
find_package(Threads REQUIRED)
add_library(ally_audio src/core/engine.cpp src/core/c_api.cpp)
add_library(Ally::Audio ALIAS ally_audio)
target_compile_features(ally_audio PUBLIC cxx_std_23)
target_include_directories(ally_audio PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(ally_audio PRIVATE Threads::Threads)
option(ALLY_AUDIO_ALSA "Build native Linux ALSA output" ON)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND ALLY_AUDIO_ALSA)
    find_package(ALSA REQUIRED)
    target_sources(ally_audio PRIVATE src/backend/alsa.cpp)
    target_link_libraries(ally_audio PRIVATE ALSA::ALSA)
else()
    target_sources(ally_audio PRIVATE src/backend/unsupported.cpp)
endif()
if(BUILD_TESTING)
    add_executable(audio_tests tests/engine_tests.cpp)
    target_link_libraries(audio_tests PRIVATE Ally::Audio)
    add_test(NAME audio.engine COMMAND audio_tests)
    add_executable(audio_c_test tests/c_api_test.c)
    target_link_libraries(audio_c_test PRIVATE Ally::Audio)
    set_target_properties(audio_c_test PROPERTIES LINKER_LANGUAGE CXX)
    add_test(NAME audio.c_abi COMMAND audio_c_test)
endif()
install(TARGETS ally_audio EXPORT AllyAudioTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT AllyAudioTargets NAMESPACE Ally:: DESTINATION lib/cmake/AllyAudio)
set_target_properties(ally_audio PROPERTIES EXPORT_NAME Audio)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/AllyAudioConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/AllyAudioConfig.cmake" INSTALL_DESTINATION lib/cmake/AllyAudio)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/AllyAudioConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/AllyAudioConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/AllyAudioConfigVersion.cmake" DESTINATION lib/cmake/AllyAudio)
